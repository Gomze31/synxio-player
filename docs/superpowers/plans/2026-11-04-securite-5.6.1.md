```kotlin
package fr.synxio.player.data.repo

/**
 * Règle d'acceptation d'une mise à jour, en empreintes SHA-256 de certificats.
 *
 * Deux conditions cumulatives :
 * 1. le signataire de l'APK téléchargé est l'empreinte épinglée [PINNED_SIGNER_SHA256],
 *    celle de la clé de release privée ;
 * 2. sa lignée (APK Signature Scheme v3) contient le signataire de l'appli installée,
 *    ce qui est aussi la règle d'Android. Une appli encore signée par l'ancienne clé
 *    accepte donc la première version signée par la nouvelle.
 *
 * Un APK signé par l'ancienne clé seule n'est jamais accepté, même par une appli qui
 * l'utilise encore : c'est le but de la rotation.
 */
object UpdateSignatures {

    /**
     * Empreinte SHA-256 du certificat de release, hexadécimal sans séparateur.
     * Vide tant que la clé n'est pas générée : aucune mise à jour n'est alors acceptée.
     */
    const val PINNED_SIGNER_SHA256 = ""

    fun compatible(
        installedCurrent: Set<String>,
        downloadedCurrent: Set<String>,
        downloadedLineage: List<String>,
        pinned: String = PINNED_SIGNER_SHA256,
    ): Boolean {
        if (pinned.isBlank()) return false
        val downloaded = downloadedCurrent.singleOrNull() ?: return false
        if (!downloaded.equals(pinned, ignoreCase = true)) return false
        val installed = installedCurrent.singleOrNull() ?: return false
        if (installed.equals(downloaded, ignoreCase = true)) return true
        return downloadedLineage.any { it.equals(installed, ignoreCase = true) }
    }
}
```

```kotlin
/**

 * `synxio.log` avec une seule rotation vers `synxio.log.1` au-delà de [maxFileBytes].
 *
 * Volontairement synchrone : seules quelques zones de l'app écrivent ici, quelques lignes
 * par événement. Un verrou suffit, pas de dispatcher dédié.
 */
class DiagnosticLog(
    private val dir: File,
    private val maxLines: Int = 500,
    private val maxFileBytes: Long = 256L * 1024,
) {
    private val lock = Any()
    private val lines = ArrayDeque<String>(maxLines)

    private val current get() = File(dir, "synxio.log")
    private val previous get() = File(dir, "synxio.log.1")

    fun append(line: String) {
        synchronized(lock) {
            if (lines.size == maxLines) lines.removeFirst()
            lines.addLast(line)
            runCatching {
                dir.mkdirs()
                if (current.exists() && current.length() > maxFileBytes) {
                    previous.delete()
                    current.renameTo(previous)
                }
                current.appendText(line + "\n")
            }
        }
    }

    fun recent(limit: Int): List<String> = synchronized(lock) { lines.takeLast(limit) }

    /** Contenu du fichier précédent puis du courant, dans l'ordre chronologique. */
    fun snapshot(): String = synchronized(lock) {
        buildString {
            if (previous.exists()) append(previous.readText())
            if (current.exists()) append(current.readText())
        }
    }

    fun sizeBytes(): Long = synchronized(lock) {
        (if (previous.exists()) previous.length() else 0L) + (if (current.exists()) current.length() else 0L)
    }

    fun clear() {
        synchronized(lock) {
            lines.clear()
            previous.delete()
            current.delete()
        }
    }
}
```

```kotlin
/**
 * Diagnostic local : journal des zones sensibles, rapports de crash, export par partage.
 * Rien ne quitte l'appareil sans un geste de l'utilisateur.
 */
@Singleton
class Diagnostics @Inject constructor(
    @ApplicationContext private val context: Context,
) {
    enum class Level { D, I, W, E }

    data class Summary(val logBytes: Long, val crashCount: Int)

    private val dir = File(context.filesDir, "diagnostics")
    private val log = DiagnosticLog(dir)
    private val prefs = context.getSharedPreferences("synxio_diagnostics", Context.MODE_PRIVATE)
    private val timestamp = SimpleDateFormat("yyyy-MM-dd HH:mm:ss.SSS", Locale.ROOT)
    private val fileStamp = SimpleDateFormat("yyyyMMdd-HHmmss", Locale.ROOT)

    fun log(tag: String, level: Level, message: String, throwable: Throwable? = null) {
        when (level) {
            Level.D -> Log.d(tag, message, throwable)
            Level.I -> Log.i(tag, message, throwable)
            Level.W -> Log.w(tag, message, throwable)
            Level.E -> Log.e(tag, message, throwable)
        }
        val line = buildString {
            append(timestamp.format(Date())).append(' ').append(level.name).append('/').append(tag)
            append(": ").append(message)
            if (throwable != null) append('\n').append(Log.getStackTraceString(throwable))
        }
        log.append(line)
    }

    /** À appeler une fois, depuis `Application.onCreate`. Le handler précédent est conservé. */
    fun installCrashHandler() {
        val previous = Thread.getDefaultUncaughtExceptionHandler()
        Thread.setDefaultUncaughtExceptionHandler { thread, error ->
            runCatching { writeCrashReport(thread, error) }
            previous?.uncaughtException(thread, error)
        }
    }

    private fun writeCrashReport(thread: Thread, error: Throwable) {
        dir.mkdirs()
        val runtime = Runtime.getRuntime()
        val header = linkedMapOf(
            "Version" to "${BuildConfig.VERSION_NAME} (${BuildConfig.VERSION_CODE})",
            "Appareil" to "${Build.MANUFACTURER} ${Build.MODEL}",
            "Android" to "${Build.VERSION.RELEASE} (API ${Build.VERSION.SDK_INT})",
            "Mémoire libre" to "${(runtime.maxMemory() - runtime.totalMemory() + runtime.freeMemory()) / (1024 * 1024)} Mo",
            "Date" to timestamp.format(Date()),
        )
        val report = CrashReportFormatter.format(header, thread.name, error, log.recent(100))
        File(dir, "crash-${fileStamp.format(Date())}.txt").writeText(report)
        crashReports().drop(MAX_CRASH_REPORTS).forEach { it.delete() }
    }

    /** Rapports du plus récent au plus ancien. */
    private fun crashReports(): List<File> =
        dir.listFiles { f -> f.name.startsWith("crash-") && f.name.endsWith(".txt") }
            .orEmpty()
            .sortedByDescending { it.name }

    fun hasUnseenCrash(): Boolean {
        val latest = crashReports().firstOrNull() ?: return false
        return latest.lastModified() > prefs.getLong(KEY_CRASH_SEEN, 0L)
    }

    fun markCrashesSeen() {
        prefs.edit().putLong(KEY_CRASH_SEEN, System.currentTimeMillis()).apply()
    }

    fun summary(): Summary = Summary(logBytes = log.sizeBytes(), crashCount = crashReports().size)

    /**
     * Concatène rapports et journal dans le cache partageable, et prépare l'intent de
     * partage. `null` si rien n'a pu être écrit.
     */
    fun exportIntent(): Intent? = runCatching {
        val out = File(context.cacheDir, "diagnostics").apply { mkdirs() }
        val file = File(out, "synxio-diagnostic-${fileStamp.format(Date())}.txt")
        file.writeText(buildString {
            appendLine("Synxio Player ${BuildConfig.VERSION_NAME} (${BuildConfig.VERSION_CODE})")
            appendLine("${Build.MANUFACTURER} ${Build.MODEL} — Android ${Build.VERSION.RELEASE} (API ${Build.VERSION.SDK_INT})")
            appendLine()
            crashReports().forEach { report ->
                appendLine("##### ${report.name}")
                appendLine(report.readText())
                appendLine()
            }
            appendLine("##### Journal")
            append(log.snapshot())
        })
        val uri = FileProvider.getUriForFile(context, "${context.packageName}.fileprovider", file)
        Intent(Intent.ACTION_SEND).apply {
            type = "text/plain"
            putExtra(Intent.EXTRA_STREAM, uri)
            putExtra(Intent.EXTRA_SUBJECT, "Diagnostic Synxio Player")
            addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
        }
    }.onFailure { log("Diagnostics", Level.W, "Export du diagnostic impossible", it) }.getOrNull()

    fun clear() {
        log.clear()
        crashReports().forEach { it.delete() }
        File(context.cacheDir, "diagnostics").deleteRecursively()
        markCrashesSeen()
    }

    private companion object {
        const val MAX_CRASH_REPORTS = 5
        const val KEY_CRASH_SEEN = "crash_seen_at"
    }
}
```

```kotlin
package fr.synxio.player.core.prefs

import java.util.Base64
import javax.crypto.Cipher
import javax.crypto.SecretKey
import javax.crypto.spec.GCMParameterSpec

/**
 * AES-256-GCM avec IV aléatoire de 12 octets, stocké devant le chiffré, le tout en
 * Base64. Indépendant d'Android : la clé vient de l'appelant (Keystore en production,
 * `KeyGenerator` en test).
 */
class SecretCodec(private val key: SecretKey) {

    fun encrypt(plain: String): String {
        val cipher = Cipher.getInstance(TRANSFORMATION)
        cipher.init(Cipher.ENCRYPT_MODE, key)
        val iv = cipher.iv
        val encrypted = cipher.doFinal(plain.toByteArray(Charsets.UTF_8))
        return Base64.getEncoder().encodeToString(iv + encrypted)
    }

    fun decrypt(stored: String): String {
        val bytes = Base64.getDecoder().decode(stored)
        require(bytes.size > IV_BYTES) { "Secret chiffré trop court" }
        val iv = bytes.copyOfRange(0, IV_BYTES)
        val encrypted = bytes.copyOfRange(IV_BYTES, bytes.size)
        val cipher = Cipher.getInstance(TRANSFORMATION)
        cipher.init(Cipher.DECRYPT_MODE, key, GCMParameterSpec(TAG_BITS, iv))
        return String(cipher.doFinal(encrypted), Charsets.UTF_8)
    }

    private companion object {
        const val TRANSFORMATION = "AES/GCM/NoPadding"
        const val IV_BYTES = 12
        const val TAG_BITS = 128
    }
}
```

```kotlin
import android.content.Context
import android.security.keystore.KeyGenParameterSpec
import android.security.keystore.KeyProperties
import dagger.hilt.android.qualifiers.ApplicationContext
import fr.synxio.player.core.diag.Diagnostics

import kotlinx.coroutines.flow.SharedFlow
import java.security.KeyStore
import java.util.concurrent.ConcurrentHashMap
import javax.crypto.KeyGenerator
import javax.crypto.SecretKey
import javax.inject.Inject
import javax.inject.Singleton

/**
 * Secrets de l'utilisateur (session Last.fm, webhook Discord), chiffrés par une clé qui
 * ne sort pas du Keystore matériel. Le fichier `synxio_secrets.xml` est exclu de la
 * sauvegarde Android : après une restauration, ces valeurs sont à ressaisir.
 *
 * Jamais de repli en clair : si le Keystore est indisponible, on lit vide et on refuse
 * d'écrire.
 */
@Singleton
class SecretStore @Inject constructor(
    @ApplicationContext context: Context,
    private val diagnostics: Diagnostics,
) {
    enum class Key(val prefName: String) {
        DISCORD_WEBHOOK_URL("discord_webhook_url"),
        LASTFM_SESSION("lastfm_session"),
    }

    private val prefs = context.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
    private val cache = ConcurrentHashMap<Key, String>()

    private val _changes = MutableSharedFlow<Unit>(replay = 1, extraBufferCapacity = 1).apply { tryEmit(Unit) }
    /** Émet à chaque écriture ; replay = 1 pour que `combine` démarre sans attendre. */
    val changes: SharedFlow<Unit> = _changes

    private val codec: SecretCodec? by lazy {
        runCatching { SecretCodec(loadOrCreateKey()) }
            .onFailure { diagnostics.log(TAG, Diagnostics.Level.E, "Keystore indisponible : secrets inaccessibles", it) }
            .getOrNull()
    }

    fun get(key: Key): String? {
        cache[key]?.let { return it }
        val stored = prefs.getString(key.prefName, null) ?: return null
        val codec = codec ?: return null
        return runCatching { codec.decrypt(stored) }
            .onFailure { diagnostics.log(TAG, Diagnostics.Level.W, "Secret ${key.name} illisible, ignoré", it) }
            .getOrNull()
            ?.also { cache[key] = it }
    }

    /** `null` ou vide supprime le secret. Lève `IllegalStateException` si le Keystore est indisponible. */
    fun put(key: Key, value: String?) {
        if (value.isNullOrBlank()) {
            prefs.edit().remove(key.prefName).apply()
            cache.remove(key)
        } else {
            val codec = codec ?: throw IllegalStateException("Keystore indisponible : impossible d'enregistrer ${key.name}")
            prefs.edit().putString(key.prefName, codec.encrypt(value)).apply()
            cache[key] = value
        }
        _changes.tryEmit(Unit)
    }

    private fun loadOrCreateKey(): SecretKey {
        val keyStore = KeyStore.getInstance(ANDROID_KEYSTORE).apply { load(null) }
        (keyStore.getEntry(ALIAS, null) as? KeyStore.SecretKeyEntry)?.let { return it.secretKey }
        val spec = KeyGenParameterSpec.Builder(ALIAS, KeyProperties.PURPOSE_ENCRYPT or KeyProperties.PURPOSE_DECRYPT)
            .setBlockModes(KeyProperties.BLOCK_MODE_GCM)
            .setEncryptionPaddings(KeyProperties.ENCRYPTION_PADDING_NONE)
            .setKeySize(256)
            .build()
        return KeyGenerator.getInstance(KeyProperties.KEY_ALGORITHM_AES, ANDROID_KEYSTORE)
            .apply { init(spec) }
            .generateKey()
    }

        const val TAG = "SecretStore"
        const val PREFS = "synxio_secrets"
        const val ANDROID_KEYSTORE = "AndroidKeyStore"
        const val ALIAS = "synxio.secrets"
```

```kotlin
/**
 * Lit le fichier `SHA2-256SUMS` publié avec chaque release de yt-dlp : une ligne par
 * fichier, `<sha256 hex>  <nom>` (le nom peut être précédé de `*`).
 */
object ShaSums {

    private val hex64 = Regex("^[0-9a-fA-F]{64}$")

    fun digestFor(sums: String, fileName: String): String? =
        sums.lineSequence()
            .map { it.trim() }
            .filter { it.isNotEmpty() }
            .mapNotNull { line ->
                val parts = line.split(Regex("\\s+"), limit = 2)
                if (parts.size != 2) return@mapNotNull null
                val name = parts[1].removePrefix("*")
                if (name != fileName) return@mapNotNull null
                parts[0].takeIf { hex64.matches(it) }?.lowercase()
            }
            .firstOrNull()
}
```

```kotlin
package fr.synxio.player.data.repo.ytdlp

import java.io.File
import java.security.MessageDigest

/**
 * Mise à jour de yt-dlp avec filet : copie de l'ancien binaire, mise à jour par la
 * bibliothèque, vérification du SHA-256 contre les sommes publiées de la release,
 * restauration au moindre doute.
 *
 * Ce que ça protège : asset substitué, téléchargement tronqué, release renommée.
 * Ce que ça ne protège pas : une release yt-dlp elle-même compromise, puisque les sommes
 * viennent de la même release.
 *
 * Les opérations Android (bibliothèque, réseau) sont injectées pour rester testable.
 */
class YtDlpUpdater(
    private val binary: File,
    private val runLibraryUpdate: () -> LibraryOutcome,
    private val installedVersion: () -> String?,
    private val fetchSums: (version: String) -> String?,
) {
    enum class LibraryOutcome { UPDATED, ALREADY_UP_TO_DATE }

    sealed interface Result {
        data class Updated(val version: String) : Result
        data object AlreadyUpToDate : Result
        data class Rejected(val reason: String) : Result
    }

    fun update(): Result {
        if (!binary.isFile) return Result.Rejected("binaire yt-dlp introuvable")
        val backup = File(binary.parentFile, "${binary.name}.bak")
        binary.copyTo(backup, overwrite = true)

        val outcome = try {
            runLibraryUpdate()
        } catch (e: Throwable) {
            restore(backup)
            return Result.Rejected("mise à jour échouée : ${e.message ?: e.javaClass.simpleName}")
        }
        if (outcome == LibraryOutcome.ALREADY_UP_TO_DATE) {
            backup.delete()
            return Result.AlreadyUpToDate
        }

        val version = installedVersion()?.takeIf { it.isNotBlank() }
            ?: return reject(backup, "version installée inconnue")
        val sums = fetchSums(version)
            ?: return reject(backup, "sommes de contrôle inaccessibles pour $version")
        val expected = ShaSums.digestFor(sums, binary.name)
            ?: return reject(backup, "aucune somme publiée pour ${binary.name} ($version)")
        val actual = sha256(binary)
        return if (actual == expected) {
            backup.delete()
            Result.Updated(version)
        } else {
            reject(backup, "somme de contrôle différente ($version)")
        }
    }

    private fun reject(backup: File, reason: String): Result {
        restore(backup)
        return Result.Rejected(reason)
    }

    private fun restore(backup: File) {
        if (backup.isFile) {
            backup.copyTo(binary, overwrite = true)
            backup.delete()
        }
    }

    private fun sha256(file: File): String {
        val digest = MessageDigest.getInstance("SHA-256")
        file.inputStream().use { input ->
            val buffer = ByteArray(64 * 1024)
            while (true) {
                val read = input.read(buffer)
                if (read < 0) break
                digest.update(buffer, 0, read)
            }
        }
        return digest.digest().joinToString("") { "%02x".format(it) }
    }
}
```

```kotlin
    suspend fun ytDlpVersion(): String? = withContext(Dispatchers.IO) {
        runCatching {
            YoutubeDL.getInstance().init(context)
            YoutubeDL.getInstance().version(context)

    /**
     * Mise à jour manuelle et vérifiée de yt-dlp. Voir [YtDlpUpdater] pour la règle ; ici
     * seulement le câblage : chemin du binaire de la bibliothèque, appel de sa mise à
     * jour, téléchargement des sommes de la release.
     */
    suspend fun updateYtDlp(): YtDlpUpdater.Result = withContext(Dispatchers.IO) {
        runCatching { YoutubeDL.getInstance().init(context) }
            .onFailure { return@withContext YtDlpUpdater.Result.Rejected("yt-dlp n'a pas pu démarrer") }
        val binary = File(context.noBackupFilesDir, "youtubedl-android/yt-dlp/yt-dlp")
        val updater = YtDlpUpdater(
            binary = binary,
            runLibraryUpdate = {
                when (YoutubeDL.getInstance().updateYoutubeDL(context, YoutubeDL.UpdateChannel._STABLE)) {
                    YoutubeDL.UpdateStatus.DONE -> YtDlpUpdater.LibraryOutcome.UPDATED
                    else -> YtDlpUpdater.LibraryOutcome.ALREADY_UP_TO_DATE
                }
            },
            installedVersion = { YoutubeDL.getInstance().version(context) },
            fetchSums = ::fetchShaSums,
        )
        val result = ytDlpLock.withLock { updater.update() }
        when (result) {
            is YtDlpUpdater.Result.Updated -> diagnostics.log(TAG, Diagnostics.Level.I, "yt-dlp mis à jour et vérifié : ${result.version}")
            is YtDlpUpdater.Result.Rejected -> diagnostics.log(TAG, Diagnostics.Level.W, "yt-dlp : mise à jour rejetée, ancien binaire restauré (${result.reason})")
            YtDlpUpdater.Result.AlreadyUpToDate -> diagnostics.log(TAG, Diagnostics.Level.I, "yt-dlp déjà à jour")
        }
        result
    }

    private fun fetchShaSums(version: String): String? = runCatching {
        val url = "https://github.com/yt-dlp/yt-dlp/releases/download/$version/SHA2-256SUMS"
        http.newCall(Request.Builder().url(url).build()).execute().use { response ->
            if (!response.isSuccessful) return@runCatching null
            response.body.string()
        }
    }.onFailure { diagnostics.log(TAG, Diagnostics.Level.W, "SHA2-256SUMS inaccessible", it) }.getOrNull()
```
