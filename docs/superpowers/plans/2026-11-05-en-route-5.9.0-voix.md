**Goal:** « Ok Google, mets-moi du rock des années 90 sur Synxio » : l'Assistant et l'écran de la voiture comprennent des phrases entières (verbe, déterminant, période, mélange, mix, reprise, radio, favoris, souvenirs), et la recherche à l'écran groupe aussi les mixs, les livres et podcasts, avec une tuile « Tout lire » pour un genre ou une période.

**Architecture:** `VoiceInterpreter` (Kotlin pur, `playback/VoiceInterpreter.kt`) transforme la requête normalisée et le focus Assistant en `VoiceIntent` (action, sujet, nom, période, mélange, texte restant). `BrowseTree.resolveVoice` résout l'intention par sujet avec les règles existantes, filtre par période, puis replie sur les favoris. `BrowseTree.search` ajoute les groupes « Mixs » et « Livres et podcasts », met les correspondances exactes en tête et une tuile « Tout lire » (`FOLDER_PLAYABLE`, identifiant `SEARCH_ALL_PREFIX`) pour un genre ou une période. `BrowseQueue` gagne un champ `radio` pour qu'une demande vocale lance une webradio.

**Tech Stack:** Kotlin 2.4, `normalizeForSearch` / `fuzzyScore` existants (`core/util/Format.kt`), JUnit 4.

```kotlin
package fr.synxio.player.playback

/**
 * Ce qu'une phrase dictée demande : une action, un sujet si la phrase le dit, un nom,
 * une période, un mélange. [text] est le reste de la phrase sans verbe ni mélange,
 * période comprise : une playlist nommée « Années 80 » doit pouvoir y correspondre.
 */
data class VoiceIntent(
    val action: Action,
    val subject: Subject?,
    val name: String?,
    val years: IntRange?,
    val shuffle: Boolean,
    val text: String?,
) {
    enum class Action { PLAY, SHUFFLE, RESUME_BOOK, RESUME_LAST }
    enum class Subject { ARTIST, ALBUM, PLAYLIST, GENRE, TITLE, RADIO, MIX, FAVORITES, MEMORY }
}

/**
 * Grammaire par motifs des demandes vocales en français, sur une requête déjà passée
 * par `normalizeForSearch` (minuscules, sans accents). Pur : ni Android, ni bibliothèque.
 *
 * Ordre : verbes, mélange, reprise, favoris, souvenirs, mixs, radio, période, déterminants.
 * Le focus de l'Assistant, s'il est là, fixe le sujet et le nom ; la grammaire ne lui
 * ajoute que la période et le mélange.
 */
object VoiceInterpreter {

    fun interpret(query: String, request: VoiceRequest, currentYear: Int): VoiceIntent {
        var text = query.replace('’', '\'').trim().replace(SPACES, " ")
        text = text.replace(VERBS, "")

        val shuffle = SHUFFLE.containsMatchIn(text)
        text = text.replace(SHUFFLE, " ").trim().replace(SPACES, " ")
        if (text in FILLERS) text = ""

        val focused = focusedSubject(request)
        if (focused == null) {
            if (RESUME_LAST.matches(text)) return VoiceIntent(VoiceIntent.Action.RESUME_LAST, null, null, null, shuffle, null)
            if (RESUME_BOOK.matches(text)) return VoiceIntent(VoiceIntent.Action.RESUME_BOOK, null, null, null, shuffle, null)
            if (FAVORITES.matches(text)) return VoiceIntent(VoiceIntent.Action.PLAY, VoiceIntent.Subject.FAVORITES, null, null, shuffle, null)
            if (MEMORY.matches(text)) return VoiceIntent(VoiceIntent.Action.PLAY, VoiceIntent.Subject.MEMORY, null, null, shuffle, null)
            if (MIX_DAILY.matches(text)) return VoiceIntent(VoiceIntent.Action.PLAY, VoiceIntent.Subject.MIX, "daily", null, shuffle, null)
            if (MIX_DISCOVERY.matches(text)) return VoiceIntent(VoiceIntent.Action.PLAY, VoiceIntent.Subject.MIX, "discovery", null, shuffle, null)
            MIX_OF.matchEntire(text)?.let { return VoiceIntent(VoiceIntent.Action.PLAY, VoiceIntent.Subject.MIX, it.groupValues[1].trim(), null, shuffle, null) }
            (RADIO_BEFORE.matchEntire(text) ?: RADIO_AFTER.matchEntire(text))?.let {
                return VoiceIntent(VoiceIntent.Action.PLAY, VoiceIntent.Subject.RADIO, it.groupValues[1].trim(), null, shuffle, null)
            }
        }

        // Le reste, déterminant de tête retiré : c'est le texte que l'appelant compare aux playlists.
        val remainder = stripLeadingDeterminer(text)
        val (years, afterPeriod) = period(remainder, currentYear)
        var name: String? = stripLeadingDeterminer(afterPeriod.replace(TRAILING_DETERMINER, "").trim()).trim()
        if (name.isNullOrEmpty() || name in FILLERS) name = null

        if (focused != null) {
            val (subject, focusName) = focused
            return VoiceIntent(VoiceIntent.Action.PLAY, subject, focusName ?: name, years, shuffle, remainder.ifEmpty { null })
        }
        val action = if (shuffle && name == null && years == null) VoiceIntent.Action.SHUFFLE else VoiceIntent.Action.PLAY
        return VoiceIntent(action, null, name, years, shuffle, remainder.ifEmpty { null })
    }

    private fun focusedSubject(request: VoiceRequest): Pair<VoiceIntent.Subject, String?>? = when (request.focus) {
        VoiceRequest.FOCUS_ARTIST -> VoiceIntent.Subject.ARTIST to request.artist
        VoiceRequest.FOCUS_ALBUM -> VoiceIntent.Subject.ALBUM to request.album
        VoiceRequest.FOCUS_GENRE -> VoiceIntent.Subject.GENRE to request.genre
        VoiceRequest.FOCUS_PLAYLIST -> VoiceIntent.Subject.PLAYLIST to request.playlist
        VoiceRequest.FOCUS_TITLE -> VoiceIntent.Subject.TITLE to request.title
        else -> null
    }

    /** Période trouvée dans [text], et le texte sans elle. */
    private fun period(text: String, currentYear: Int): Pair<IntRange?, String> {
        fun decade(raw: String): IntRange {
            val n = raw.toInt()
            val start = when {
                n >= 1000 -> n / 10 * 10
                n >= 30 -> 1900 + n / 10 * 10
                else -> 2000 + n / 10 * 10
            }
            return start..start + 9
        }
        fun cut(m: MatchResult) = (text.substring(0, m.range.first) + " " + text.substring(m.range.last + 1)).trim().replace(SPACES, " ")

        RANGE.find(text)?.let { return it.groupValues[1].toInt()..it.groupValues[2].toInt() to cut(it) }
        DECADE_WORD.find(text)?.let { return decade(it.groupValues[1]) to cut(it) }
        DECADE_SUFFIX.find(text)?.let { return decade(it.groupValues[1]) to cut(it) }
        DECADE_NAME.find(text)?.let { return DECADES.getValue(it.groupValues[1]) to cut(it) }
        BEFORE.find(text)?.let { return 1900..it.groupValues[1].toInt() - 1 to cut(it) }
        AFTER.find(text)?.let { return it.groupValues[1].toInt() + 1..currentYear to cut(it) }
        SINCE.find(text)?.let { return it.groupValues[1].toInt()..currentYear to cut(it) }
        SINGLE_YEAR.find(text)?.let { return it.groupValues[1].toInt()..it.groupValues[1].toInt() to cut(it) }
        RECENT.find(text)?.let { return currentYear - 2..currentYear to cut(it) }
        return null to text
    }

    /** Retire « du », « de la », « un peu de »… devant un mot ; jamais devant un nombre (« de 2015 »). */
    private fun stripLeadingDeterminer(text: String): String = text.replace(LEADING_DETERMINER, "").trim()

    private val SPACES = Regex("\\s+")
    private val VERBS = Regex(
        "^(?:mets|met|joue|jouer|lance|lancer|ecoute|ecouter|passe|fais(?:[\\s-]+moi)?[\\s-]+ecouter|je veux ecouter|je voudrais ecouter|tu peux (?:me )?(?:mettre|jouer|lancer))(?:[\\s-]+(?:moi|nous))?\\s+"
    )
    private val SHUFFLE = Regex("\\b(?:en aleatoire|au hasard|en melange|melangee?s?|aleatoire|shuffle)\\b")
    private val FILLERS = setOf("musique", "de la musique", "lecture", "quelque chose", "un truc", "des titres", "des chansons", "un morceau", "des morceaux")

    private val RESUME_LAST = Regex(
        "(?:(?:reprends?|reprendre|continue|remets?)\\s+)?(?:ce que j'?ecoutais|ce que j'?etais en train d'?ecouter|ma derniere file|la derniere file|ma file(?: d'?attente)?|la file(?: d'?attente)?)"
    )
    private val RESUME_BOOK = Regex(
        "(?:continue|reprends?|reprendre|remets?)(?:\\s+(?:mon livre(?: audio)?|mon podcast|mon bouquin|la lecture|la ou j'?en etais|ou j'?en etais))?" +
            "|mon livre(?: audio)?|mon podcast|la ou j'?en etais|ou j'?en etais"
    )
    private val FAVORITES = Regex("(?:tous mes |mes |les )?(?:titres |chansons |morceaux )?(?:favoris|favorites|preferes|preferees|coups de coeur)")
    private val MEMORY = Regex("(?:mes )?souvenirs(?: musicaux)?|ce jour[- ]la|il y a (?:un|deux) ans?")
    private val MIX_DAILY = Regex("(?:mon |le |ton )?mix du jour|mon mix")
    private val MIX_DISCOVERY = Regex("(?:ma |la |ta )?decouverte du jour")
    private val MIX_OF = Regex("(?:le |un |mon )mix (?:de |d'|du |des )?(.+)")
    private val RADIO_BEFORE = Regex("(?:la |une )?(?:web)?radio (.+)")
    private val RADIO_AFTER = Regex("(.+) en (?:web)?radio")

    private val RANGE = Regex("\\bde (\\d{4}) a (\\d{4})\\b")
    private val DECADE_WORD = Regex("(?:\\b(?:les|des|du|de la|de l'|d')\\s+)?\\bannees? (\\d{4}|\\d{2})\\b")
    private val DECADE_SUFFIX = Regex("\\b(\\d{4}|\\d{2})s\\b")
    private val DECADE_NAME = Regex("\\b(sixties|seventies|eighties|nineties)\\b")
    private val DECADES = mapOf("sixties" to 1960..1969, "seventies" to 1970..1979, "eighties" to 1980..1989, "nineties" to 1990..1999)
    private val BEFORE = Regex("\\bavant (\\d{4})\\b")
    private val AFTER = Regex("\\bapres (\\d{4})\\b")
    private val SINCE = Regex("\\bdepuis (\\d{4})\\b")
    private val SINGLE_YEAR = Regex("\\b(?:de|en) (\\d{4})\\b")
    private val RECENT = Regex("\\b(?:recents?|recentes?|nouveautes|nouveaux?|nouvelles?)\\b")

    private val LEADING_DETERMINER = Regex("^(?:un peu de |un peu d'|du |de la |de l'|des |de |d'|le |la |les |l'|un |une )(?=[a-z])")
    private val TRAILING_DETERMINER = Regex("\\s+(?:du|de la|de l'|des|de|d')$")
}
```

    /**
     * Résout « Joue … » dicté à l'Assistant ou tapé sur l'écran.
     *
     * [VoiceInterpreter] lit la phrase ; chaque sujet se résout avec les règles de nom,
     * puis la période filtre. Sans sujet, les règles s'enchaînent : artiste, album,
     * playlist, genre, titres. Repli : titres de la période, puis favoris au hasard.
     * Null seulement sans permission ou sans bibliothèque.
     */
    fun resolveVoice(request: VoiceRequest): BrowseQueue? {
        if (!source.hasPermission || source.songs.isEmpty()) return null
        val intent = VoiceInterpreter.interpret(request.query.normalizeForSearch(), request, currentYear())
        val years = intent.years
        fun queue(songs: List<Song>) = BrowseQueue(if (intent.shuffle) shuffle(songs) else songs)

        // Une playlist qui porte le nom entier de la demande gagne (« Années 80 »).
        intent.text?.let { best(source.playlists.filter { p -> p.songs.isNotEmpty() }, it, EXACT_SCORE) { p -> p.name } }
            ?.let { return queue(it.songs) }

        when (intent.action) {
            VoiceIntent.Action.RESUME_BOOK -> continueListening().firstOrNull()?.let { return it.queue }
            VoiceIntent.Action.RESUME_LAST -> source.lastQueue?.takeIf { it.songs.isNotEmpty() }?.let { return it }
            VoiceIntent.Action.SHUFFLE -> return BrowseQueue(shuffle(source.songs))
            VoiceIntent.Action.PLAY -> Unit
        }

        val name = intent.name
        when (intent.subject) {
            VoiceIntent.Subject.ARTIST -> name?.let(::bestArtist)?.let { return queue(it.songs.inYears(years)) }
            VoiceIntent.Subject.ALBUM -> name?.let { bestAlbum(it, request.artist) }?.let { return queue(it.songs.inYears(years)) }
            VoiceIntent.Subject.PLAYLIST -> name?.let(::bestPlaylist)?.let { return queue(it.songs.inYears(years)) }
            VoiceIntent.Subject.GENRE -> name?.let(::bestGenre)?.let { return BrowseQueue(shuffle(it.songs.inYears(years))) }
            VoiceIntent.Subject.TITLE -> name?.let { bestSong(it, request.artist) }?.let { return songWithAlbum(it) }
            VoiceIntent.Subject.MIX -> bestMix(name)?.let { return queue(it.songs) }
            VoiceIntent.Subject.MEMORY -> source.memories.firstOrNull { it.songs.isNotEmpty() }?.let { return queue(it.songs) }
            VoiceIntent.Subject.RADIO -> name?.let { best(source.radios, it) { r -> r.name } }?.let { return BrowseQueue(emptyList(), radio = it) }
            VoiceIntent.Subject.FAVORITES -> if (source.favorites.isNotEmpty()) return queue(source.favorites.inYears(years))
            null -> Unit
        }

        // Règles générales sur le nom, la période appliquée ensuite.
        if (name != null) {
            bestArtist(name)?.let { return queue(it.songs.inYears(years)) }
            bestAlbum(name, request.artist)?.let { return queue(it.songs.inYears(years)) }
            bestPlaylist(name)?.let { return queue(it.songs.inYears(years)) }
            bestGenre(name)?.let { return BrowseQueue(shuffle(it.songs.inYears(years))) }
            val results = source.search(name, 50)
            if (results.songs.isNotEmpty()) return queue(results.songs.inYears(years))
            results.albums.flatMap { it.songs }.takeIf { it.isNotEmpty() }?.let { return queue(it.inYears(years)) }
        }

        // Période seule, ou genre inconnu avec une période : tous les titres de la période.
        if (years != null) {
            source.songs.filter { it.year in years }.takeIf { it.isNotEmpty() }?.let { return BrowseQueue(shuffle(it)) }
        }
        // Rien demandé de précis (« mets de la musique ») : toute la bibliothèque. Sinon, y
        // compris une reprise sans rien à reprendre, les favoris au hasard.
        if (intent.action == VoiceIntent.Action.PLAY && name == null && years == null && intent.subject == null) {
            return BrowseQueue(shuffle(source.songs))
        }
        return BrowseQueue(shuffle(source.favorites.ifEmpty { source.songs }))
    }

    /** Filtre par année de sortie ; une période sans aucun titre rend la liste entière. */
    private fun List<Song>.inYears(years: IntRange?): List<Song> =
        if (years == null) this else filter { it.year in years }.ifEmpty { this }

    /** Mix du jour (`daily`, `discovery`), mix d'un artiste par son nom, sinon le premier. */
    private fun bestMix(name: String?): DailyMix? {
        val mixes = source.dailyMixes.filter { it.songs.isNotEmpty() }
        if (name == null) return mixes.firstOrNull()
        mixes.firstOrNull { it.key == name }?.let { return it }
        return best(mixes.filter { it.kind == DailyMixKind.ARTIST }, name) { it.title.removePrefix("Mix ") } ?: mixes.firstOrNull()
    }

    private fun currentYear(): Int = java.time.LocalDate.ofEpochDay(today()).year

    /**
     * Résultats de la recherche tapée ou dictée sur l'écran de la voiture, par sections.
     *
     * En tête : « Tout lire » si la demande est un genre ou une période, puis l'artiste
     * ou la playlist qui porte exactement ce nom. Ensuite trois titres, les artistes,
     * albums, playlists, mixs, livres et podcasts, radios, puis le reste des titres.
     */
    fun search(query: String): List<BrowseNode> {
        val q = query.trim()
        if (q.isEmpty() || !source.hasPermission) return emptyList()
        val results = source.search(q, SEARCH_LIMIT)
        val parent = BrowseIds.SEARCH_PREFIX + q

        val artists = results.artists.take(5).map { artistNode(it).copy(groupTitle = "Artistes") to fuzzyScore(it.name, q) }
        val playlists = matchingPlaylists(q).take(5).map { playlistNode(it).copy(groupTitle = "Playlists") to fuzzyScore(it.name, q) }
        val exact = (artists + playlists).filter { it.second >= EXACT_SCORE }.map { it.first }
        val albums = results.albums.take(8).map { albumNode(it).copy(groupTitle = "Albums") }
        val mixes = source.dailyMixes.filter { it.songs.isNotEmpty() && fuzzyScore(it.title, q) >= MATCH_SCORE }.take(3)
            .map { mixNode(it).copy(groupTitle = "Mixs") }
        val listening = books().filter { fuzzyScore(it.title, q) >= MATCH_SCORE }.take(3).map { it.copy(groupTitle = "Livres et podcasts") } +
            source.podcasts.filter { fuzzyScore(it.title.cleanTitle(), q) >= MATCH_SCORE }.take(3)
                .map { episodeNode(BrowseIds.CAT_PODCASTS, it).copy(groupTitle = "Livres et podcasts") }
        val radios = source.radios.filter { fuzzyScore(it.name, q) >= MATCH_SCORE }.take(5)
            .map { radioNode(it).copy(groupTitle = "Radios") }
        val songs = results.songs.take(SEARCH_LIMIT).map { songNode(parent, it).copy(groupTitle = "Titres") }

        val intent = VoiceInterpreter.interpret(q.normalizeForSearch(), VoiceRequest(q), currentYear())
        val playAll = if (intent.years != null || bestGenre(q) != null) {
            songsOf(BrowseIds.SEARCH_ALL_PREFIX + q)?.let { all ->
                listOf(BrowseNode(BrowseIds.SEARCH_ALL_PREFIX + q, "Tout lire", titles(all.size), NodeKind.FOLDER_PLAYABLE,
                    NodeArt.Icon(BrowseIcon.SHUFFLE), playableStyle = ChildStyle.LIST, actions = CONTAINER_ACTIONS))
            }.orEmpty()
        } else emptyList()

        val ordered = playAll + exact + songs.take(3) + artists.map { it.first } + albums + playlists.map { it.first } +
            mixes + listening + radios + songs.drop(3)
        return ordered.distinctBy { it.id }.take(SEARCH_LIMIT)
    }
